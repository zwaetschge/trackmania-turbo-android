cmake_minimum_required(VERSION 3.18)
project(tmhost C CXX)
set(TM_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/..)
set(PCRECOMP ${TM_ROOT}/third_party/pcrecomp)
set(GEN_DIR "" CACHE PATH "directory with generated recomp_*.c")
set(STUB OFF CACHE BOOL "stub dispatch")
set(DXVK_SOURCE ${TM_ROOT}/work/dxvk-source CACHE PATH "prepared DXVK")
file(GLOB RT_SOURCES CONFIGURE_DEPENDS ${TM_ROOT}/runtime/src/*.c)
set(COMMON_DEFS RECOMP_STRICT_ICALL RECOMP_X87_WINDOW "RECOMP_TLS=__thread" RECOMP_CPU_STRUCT RECOMP_FAST_DISPATCH)
# DXVK backend (d3d11+dxgi) + WSI + the guest<->native bridge
add_subdirectory(${TM_ROOT}/gfx/dxvk dxvk)
add_library(tm_bridge STATIC ${TM_ROOT}/gfx/d3d11_bridge.cpp)
target_include_directories(tm_bridge PRIVATE ${TM_ROOT}/runtime/include ${PCRECOMP}/runtime/recomp32)
target_link_libraries(tm_bridge PRIVATE tm_dxvk)
target_compile_definitions(tm_bridge PRIVATE ${COMMON_DEFS})
target_compile_options(tm_bridge PRIVATE -O1 -g -w -fno-strict-aliasing)
if(STUB)
  set(GEN_SOURCES ${TM_ROOT}/work/stubgen/stub_dispatch.c)
else()
  file(GLOB GEN_SOURCES CONFIGURE_DEPENDS ${GEN_DIR}/recomp_*.c ${TM_ROOT}/gen/extra/extra_recomp_*.c
       ${TM_ROOT}/gen/am/am_recomp_*.c ${TM_ROOT}/gen/wp/wp_recomp_*.c)
  add_library(tm_guest OBJECT ${GEN_SOURCES})
  target_include_directories(tm_guest PRIVATE ${PCRECOMP}/runtime/recomp32 ${GEN_DIR} ${TM_ROOT}/gen/extra)
  target_compile_definitions(tm_guest PRIVATE ${COMMON_DEFS} RECOMP_LOCAL_REGS "RECOMP_MEM_QUAL=")
  set(GUEST_OPT "-O0" CACHE STRING "guest opt")
  target_compile_options(tm_guest PRIVATE ${GUEST_OPT} -w -fwrapv -fno-strict-aliasing -ffp-contract=off -g0
      -include ${TM_ROOT}/runtime/include/guest_prelude.h)
  set(GEN_SOURCES $<TARGET_OBJECTS:tm_guest>)
endif()
add_executable(tmhost ${TM_ROOT}/host/main_host.c ${RT_SOURCES} ${GEN_SOURCES})
target_include_directories(tmhost PRIVATE ${TM_ROOT}/runtime/include ${PCRECOMP}/runtime/recomp32)
target_compile_definitions(tmhost PRIVATE ${COMMON_DEFS})
target_compile_options(tmhost PRIVATE -O1 -g -Wall -Wextra -Wno-unused-parameter -Wno-missing-field-initializers -fno-strict-aliasing)
find_library(VK vulkan REQUIRED)
find_library(X11LIB X11 REQUIRED)
# OpenAL (native OpenAL Soft behind the guest's OpenAL32.dll)
add_subdirectory(${TM_ROOT}/audio audio)
target_compile_definitions(tmhost PRIVATE TM_HAVE_OPENAL)
target_link_libraries(tmhost PRIVATE tm_bridge tm_dxvk tm_audio ${VK} ${X11LIB} pthread m stdc++ dl)
# Native VP8 decoder (runtime/src/video.c): optional on the host, for testing the hook.
#   mkdir -p work/libvpx-host && cd work/libvpx-host && ../../third_party/libvpx/configure --target=generic-gnu
#     --disable-examples --disable-tools --disable-docs --disable-unit-tests --disable-vp9 --disable-vp8-encoder
#     --disable-webm-io --disable-libyuv --enable-pic && make -j8 libvpx.a
set(TM_VPX_LIB ${TM_ROOT}/work/libvpx-host/libvpx.a)
if(EXISTS ${TM_VPX_LIB})
  target_compile_definitions(tmhost PRIVATE TM_HAVE_VPX)
  target_include_directories(tmhost PRIVATE ${TM_ROOT}/third_party/libvpx ${TM_ROOT}/work/libvpx-host)
  target_link_libraries(tmhost PRIVATE ${TM_VPX_LIB})
endif()
set_target_properties(tmhost PROPERTIES LINKER_LANGUAGE CXX)
